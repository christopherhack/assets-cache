local function httpRequest(options)
    local fn = (syn and syn.request) or (http and http.request) or http_request or request
    if not fn then error("Executor does not support HTTP requests.") end
    return fn(options)
end

-- Retries (Render free/cold instances can take ~30s to wake up)
local function request(options, attempts, onRetry)
    local lastErr = "unknown error"
    for i = 1, attempts do
        local ok, res = pcall(httpRequest, options)
        if ok and type(res) == "table" and res.Body and (res.StatusCode or 0) < 500 then
            return res
        end
        lastErr = ok and ("HTTP " .. tostring(res and res.StatusCode)) or tostring(res)
        if i < attempts then
            if onRetry then onRetry(i) end
            task.wait(4)
        end
    end
    return nil, lastErr
end

local function decode(body)
    local ok, data = pcall(function() return HttpService:JSONDecode(body) end)
    return ok and type(data) == "table" and data or nil
end

local function getHWID()
    local ok, id = pcall(function()
        if gethwid then return gethwid() end
        return game:GetService("RbxAnalyticsService"):GetClientId()
    end)
    return (ok and id) and tostring(id) or "unknown"
end

local function saveKey(key)
    pcall(function() if writefile then writefile(KEY_FILE, key) end end)
end
local function loadKey()
    local g = (getgenv and getgenv().script_key) or rawget(_G, "script_key")
    if type(g) == "string" and g ~= "" then return g end
    local ok, key = pcall(function()
        if isfile and readfile and isfile(KEY_FILE) then return readfile(KEY_FILE) end
    end)
    return (ok and type(key) == "string") and key:gsub("%s+", "") or ""
end
local function forgetKey()
    pcall(function() if delfile and isfile and isfile(KEY_FILE) then delfile(KEY_FILE) end end)
end

local function guiParent()
    local ok, ui = pcall(function() return gethui and gethui() end)
    return (ok and ui) or CoreGui
end

local function tween(obj, time, style, dir, props)
    local t = TweenService:Create(obj, TweenInfo.new(time, style, dir), props)
    t:Play()
    return t
end

local function mkSound(name, id, vol)
    if not id or id == "" or id:find("YOUR_") then return nil end
    local s = Instance.new("Sound")
    s.Name = name
    s:SetAttribute("GENX", true) -- marks it so the mute system never touches it
    s.SoundId = id
    s.Volume = vol
    s.Looped = false
    s.Parent = SoundService
    return s
end
local IntroSfx = mkSound("GENXIntroSfx", INTRO_SOUND_ID, 0)
local OutroSfx = mkSound("GENXOutroSfx", OUTRO_SOUND_ID, OUTRO_VOLUME)
if OutroSfx then OutroSfx.Looped = true end -- heartbeat loops until the key is accepted

-- Game mute: zero every Sound that isn't ours, keep new ones muted, hold them at 0
local mutedVolumes = {}
local muteConns = {}

local function muteSound(s)
    if not s:IsA("Sound") or s:GetAttribute("GENX") then return end
    if mutedVolumes[s] == nil then mutedVolumes[s] = s.Volume end
    s.Volume = 0
    table.insert(muteConns, s:GetPropertyChangedSignal("Volume"):Connect(function()
        if s.Volume ~= 0 then
            mutedVolumes[s] = s.Volume -- remember what the game wanted
            s.Volume = 0
        end
    end))
end

local function muteGame()
    if not MUTE_GAME_AUDIO then return end
    pcall(function()
        for _, d in ipairs(game:GetDescendants()) do muteSound(d) end
    end)
    table.insert(muteConns, game.DescendantAdded:Connect(muteSound))
end

local function restoreGame()
    for _, c in ipairs(muteConns) do c:Disconnect() end
    table.clear(muteConns)
    if RESTORE_GAME_AUDIO then
        for s, v in pairs(mutedVolumes) do
            if s.Parent then pcall(function() s.Volume = v end) end
        end
    end
    table.clear(mutedVolumes)
end

local function playSfx(s, vol, fadeIn)
    if not s then return end
    task.spawn(function()
        pcall(function() ContentProvider:PreloadAsync({ s }) end)
        if not s.Parent or not s.IsLoaded then return end -- skip silently if the asset is blocked/private
        pcall(function() s.TimePosition = 0 s:Play() end)
        tween(s, fadeIn or 0.2, ES.Sine, ED.Out, { Volume = vol })
    end)
end

local function fadeStopSfx(s, t)
    if not s then return end
    tween(s, t, ES.Sine, ED.In, { Volume = 0 })
    task.delay(t + 0.1, function() pcall(function() s:Stop() s:Destroy() end) end)
end

ScreenGui.Destroying:Connect(function()
    restoreGame()
    if IntroSfx then pcall(function() IntroSfx:Destroy() end) end
    if OutroSfx then pcall(function() OutroSfx:Stop() OutroSfx:Destroy() end) end
end)

-- One heartbeat: soft P wave, sharp QRS spike, rounded T wave
local WAVE = {
    { 0, 0 }, { 0.16, 0 }, { 0.19, 0.05 }, { 0.22, 0.10 }, { 0.25, 0.05 }, { 0.28, 0 },
    { 0.40, 0 }, { 0.43, -0.12 }, { 0.47, 1 }, { 0.52, -0.38 }, { 0.55, 0 }, { 0.64, 0 },
    { 0.69, 0.10 }, { 0.74, 0.20 }, { 0.79, 0.10 }, { 0.84, 0 }, { 1, 0 },
}
local EcgLines = {}
local function buildEcg(y, amp, period, speed, color, alpha)
    local copies = math.ceil(CARD_W / period) + 2
    local holder = Instance.new("Frame")
    holder.AnchorPoint = Vector2.new(0, 0.5)
    holder.Position = UDim2.new(0, 0, y, 0)
    holder.Size = UDim2.fromOffset(period * copies, 2)
    holder.BackgroundTransparency = 1
    holder.ZIndex = 2
    holder.Parent = Ecg
    for c = 0, copies - 1 do
        for i = 1, #WAVE - 1 do
            local x1, y1 = (c + WAVE[i][1]) * period, -WAVE[i][2] * amp
            local x2, y2 = (c + WAVE[i + 1][1]) * period, -WAVE[i + 1][2] * amp
            local dx, dy = x2 - x1, y2 - y1
            local seg = Instance.new("Frame")
            seg.AnchorPoint = Vector2.new(0.5, 0.5)
            seg.Position = UDim2.fromOffset((x1 + x2) / 2, 1 + (y1 + y2) / 2)
            seg.Size = UDim2.fromOffset(math.sqrt(dx * dx + dy * dy) + 1.2, 1.6)
            seg.Rotation = math.deg(math.atan2(dy, dx))
            seg.BackgroundColor3 = color
            seg.BackgroundTransparency = alpha
            seg.BorderSizePixel = 0
            seg.ZIndex = 2
            seg.Parent = holder
        end
    end
    table.insert(EcgLines, { holder = holder, y = y, period = period, speed = speed,
        offset = rng:NextNumber(0, period) })
end

buildEcg(0.07, 16, 250, 55,  PURPLE, 0.60)
buildEcg(0.21, 24, 290, 80,  PINK,   0.50)

buildEcg(0.50, 28, 320, 70,  PINK,   0.45)
buildEcg(0.64, 20, 260, 95,  PURPLE, 0.60)

local Motes = {} -- a few slow, faint particles inside the card
for i = 1, 8 do
    local sz = rng:NextInteger(2, 3)
    local m = makeCircle(Ecg, UDim2.fromOffset(sz, sz), UDim2.fromScale(0.5, 0.5), 1)
    m.BackgroundColor3 = (i % 2 == 0) and PINK or PURPLE
    m.BackgroundTransparency = rng:NextNumber(0.6, 0.8)
    Motes[i] = { f = m, x = rng:NextNumber(0.05, 0.95), y = rng:NextNumber(0, 1), v = rng:NextNumber(0.012, 0.03) }
end

local ecgShow, ecgTarget, bgClock, beatLive = 0, 1, 0, false
local bgConn = RunService.RenderStepped:Connect(function(dt)
    if not KeyFrame.Visible then return end
    bgClock = bgClock + dt
    for _, m in ipairs(Motes) do
        m.y = m.y - m.v * dt
        if m.y < 0 then m.y = 1 end
        m.f.Position = UDim2.fromScale(m.x, m.y)
    end
    ecgShow = ecgShow + (ecgTarget - ecgShow) * math.min(1, dt * 5)
    for _, L in ipairs(EcgLines) do
        L.holder.Position = UDim2.new(0, -((bgClock * L.speed + L.offset) % L.period), L.y, 0)
    end
    -- lub-dub: strong beat, then a softer second one
    local ph = bgClock % 1.1
    local pulse = math.min(1, math.exp(-ph * 9) + (ph > 0.3 and 0.65 * math.exp(-(ph - 0.3) * 9) or 0))
    Ecg.GroupTransparency = 1 - ecgShow * (0.55 + 0.45 * pulse)
    EcgGlow.BackgroundTransparency = 0.9 - 0.08 * pulse
    if beatLive then
        EnterStroke.Transparency = 0.75 - 0.6 * pulse
        KeyLogoStroke.Thickness = 1.5 + 1.5 * pulse -- logo ring beats too
        if not keyFocused then KeyGlowStroke.Transparency = 0.9 - 0.3 * pulse end -- input halo beats
    end
end)

local function fadeKey(show, t)
    if not show then
        for _, item in ipairs(KeyFade) do
            local goal = {}
            for prop in pairs(item.goal) do goal[prop] = 1 end
            tween(item.inst, t, ES.Quad, ED.In, goal)
        end
        tween(KeyScale, t, ES.Quad, ED.In, { Scale = keyBase * 0.94 })
        return
    end

    -- entrance order: card, header, status, input, activate button, discord button
    local groups = {
        { 0.12, { DragBar } },
        { 0.22, { StatusLabel, StatusDot } },
        { 0.32, { KeyBox, KeyGlow, PasteBtn } },
        { 0.42, { EnterBtn } },
        { 0.52, { DiscordBtn } },
    }
    local function delayOf(inst)
        for _, g in ipairs(groups) do
            for _, root in ipairs(g[2]) do
                if inst == root or inst:IsDescendantOf(root) then return g[1] end
            end
        end
        return 0
    end

    for _, g in ipairs(groups) do
        for _, root in ipairs(g[2]) do
            local final = root.Position
            root.Position = final + UDim2.fromOffset(0, 10)
            task.delay(g[1], function() tween(root, 0.45, ES.Quint, ED.Out, { Position = final }) end)
        end
    end
    for _, item in ipairs(KeyFade) do
        local goal = {}
        for prop, value in pairs(item.goal) do
            item.inst[prop] = 1
            goal[prop] = value
        end
        task.delay(delayOf(item.inst), function()
            tween(item.inst, 0.45, ES.Quad, ED.Out, goal)
        end)
    end
    KeyScale.Scale = keyBase * 0.92
    KeyFrame.Position = UDim2.new(0.5, 0, 0.5, 22)
    tween(KeyScale, t, ES.Back, ED.Out, { Scale = keyBase })
    tween(KeyFrame, t + 0.1, ES.Quint, ED.Out, { Position = UDim2.new(0.5, 0, 0.5, 0) })
end

local function fadeOutTree(root, t)
    for _, d in ipairs(root:GetDescendants()) do
        local goal = {}
        if d:IsA("GuiObject") then goal.BackgroundTransparency = 1 end
        if d:IsA("TextLabel") then goal.TextTransparency = 1 end
        if d:IsA("ImageLabel") then goal.ImageTransparency = 1 end
        if d:IsA("UIStroke") then goal.Transparency = 1 end
        if next(goal) then
            tween(d, t, ES.Quad, ED.In, goal)
        end
    end
end

local function flash(strength, t)
    Flash.BackgroundTransparency = 1 - strength
    tween(Flash, t or 0.6, ES.Quad, ED.Out, { BackgroundTransparency = 1 })
end

-- Expanding ring that fades out
local function shockwave(startSize, endSize, t, thickness)
    local f = makeCircle(Stage, UDim2.fromOffset(startSize, startSize), CENTER_ST, 3)
    local s = Instance.new("UIStroke")
    s.Thickness = thickness or 3
    s.Color = PINK
    s.Transparency = 0.3
    s.Parent = f
    tween(f, t, ES.Quad, ED.Out, { Size = UDim2.fromOffset(endSize, endSize) })
    tween(s, t, ES.Quad, ED.Out, { Transparency = 1 })
    task.delay(t + 0.1, function() f:Destroy() end)
end

-- Drifting background particles
local function startParticles()
    for _ = 1, 28 do
        task.spawn(function()
            task.wait(rng:NextNumber(0, 5))
            local sz = rng:NextInteger(2, 5)
            local p = makeCircle(Container, UDim2.fromOffset(sz, sz), UDim2.new(0.5, 0, 1.1, 0), 1)
            p.BackgroundColor3 = rng:NextNumber() > 0.5 and PURPLE or PINK
            while introRunning and p.Parent do
                local x = rng:NextNumber(0.02, 0.98)
                local dur = rng:NextNumber(5, 10)
                p.Position = UDim2.new(x, 0, 1.05, 0)
                p.BackgroundTransparency = rng:NextNumber(0.45, 0.85)
                tween(p, dur, ES.Linear, ED.Out,
                    { Position = UDim2.new(x + rng:NextNumber(-0.06, 0.06), 0, -0.05, 0) })
                task.wait(dur)
            end
        end)
    end
end

local DECODE_CHARS = { "G", "E", "N", "X", "#", "/", "0", "1", "$", "%", "<", ">" }

local function decodeLetter(l, finalChar, dur)
    task.spawn(function()
        tween(l.label, 0.2, ES.Quad, ED.Out, { TextTransparency = 0 })
        tween(l.stroke, 0.2, ES.Quad, ED.Out, { Transparency = 0.45 })
        l.label.TextColor3 = PINK
        local t0 = os.clock()
        while os.clock() - t0 < dur do
            l.label.Text = DECODE_CHARS[rng:NextInteger(1, #DECODE_CHARS)]
            task.wait(0.04)
        end
        l.label.Text = finalChar
        l.label.TextColor3 = WHITE
        l.label.Position = UDim2.fromOffset(0, -5)
        tween(l.label, 0.25, ES.Back, ED.Out, { Position = UDim2.fromOffset(0, 0) })
    end)
end

local statusToken = 0
local function typeStatus(text)
    statusToken = statusToken + 1
    local mine = statusToken
    task.spawn(function()
        for i = 1, #text do
            if mine ~= statusToken then return end
            StageText.Text = text:sub(1, i)
            task.wait(0.018)
        end
    end)
end

local function playIntroAnimation()
    -- Wake the Render server while the intro plays (cold starts can take ~30s)
    local warmDone, warmOk = false, false
    task.spawn(function()
        local res = request({ Url = BACKEND_URL .. "/health", Method = "GET" }, 8)
        warmOk = res ~= nil
        local info = INFO or (res and decode(res.Body))
        if info then
            if info.message_id then lastMsgId = info.message_id end
            applyMessage(info.message)
        end
        warmDone = true
    end)

    muteGame()
    if OutroSfx then
        task.spawn(function() pcall(function() ContentProvider:PreloadAsync({ OutroSfx }) end) end)
    end
    playSfx(IntroSfx, INTRO_VOLUME, 0.25)

    local clock, ticksLive = 0, false
    table.insert(introConns, RunService.RenderStepped:Connect(function(dt)
        clock = clock + dt
        for _, a in ipairs(Arcs) do
            a.grad.Rotation = (a.grad.Rotation + a.speed * dt) % 360
        end
        for _, o in ipairs(Orbs) do
            local ang = o.phase + clock * o.speed
            o.frame.Position = UDim2.fromOffset(CX + math.cos(ang) * o.radius, CY + math.sin(ang) * o.radius)
        end
        TickHolder.Rotation = (clock * 5) % 360
        if ticksLive then
            local head = math.floor(clock * 16) % TICKS
            for i, tk in ipairs(Ticks) do
                local base = ((i - 1) % 5 == 0) and 0.15 or 0.6
                local d = (head - i) % TICKS
                tk.BackgroundTransparency = d < 14 and base * (d / 14) or base
            end
        end
    end))
    startParticles()

    -- Phase 1: corners + glow, scan line drops onto the logo position
    local cam = workspace.CurrentCamera
    local vp = cam and cam.ViewportSize or Vector2.new(1280, 720)
    local logoY = vp.Y / 2 + (CY - STAGE_H / 2) * StageScale.Scale

    for _, b in ipairs(Brackets) do
        tween(b, 0.8, ES.Quad, ED.Out, { BackgroundTransparency = 0.35 })
    end
    for i, g in ipairs(Glow) do
        tween(g, 1.0, ES.Sine, ED.Out, { BackgroundTransparency = glowTargets[i] })
    end
    tween(ScanBar, 0.15, ES.Quad, ED.Out, { BackgroundTransparency = 0.1 })
    tween(ScanBar, 0.6, ES.Quad, ED.In, { Position = UDim2.new(0.5, 0, 0, logoY) })
    task.wait(0.6)

    -- Phase 2: scan hits the logo: flash, pop, shockwaves, tick ring lights up
    flash(0.5, 0.6)
    tween(ScanBar, 0.7, ES.Quad, ED.Out,
        { Position = UDim2.new(0.5, 0, 0, vp.Y + 10), BackgroundTransparency = 1 })
    LogoHolder.Rotation = -30
    tween(LogoScale, 0.85, ES.Back, ED.Out, { Scale = 1 })
    tween(LogoHolder, 0.85, ES.Back, ED.Out, { Rotation = 0 })
    tween(LogoHolder, 0.4, ES.Quad, ED.Out, { BackgroundTransparency = 0 })
    tween(LogoHolderStroke, 0.5, ES.Quad, ED.Out, { Transparency = 0.2 })
    tween(Logo, 0.5, ES.Quad, ED.Out, { ImageTransparency = 0 })
    tween(LogoFallback, 0.5, ES.Quad, ED.Out, { TextTransparency = 0 })
    tween(LogoFallbackStroke, 0.5, ES.Quad, ED.Out, { Transparency = 0.3 })
    shockwave(150, 540, 1.2, 3)
    task.delay(0.18, function() shockwave(150, 420, 1.0, 2) end)

    tween(BaseStroke, 0.8, ES.Sine, ED.Out, { Transparency = 0.75 })
    for _, a in ipairs(Arcs) do
        tween(a.stroke, 0.8, ES.Sine, ED.Out, { Transparency = a.alpha })
    end
    for _, o in ipairs(Orbs) do
        tween(o.frame, 0.8, ES.Sine, ED.Out, { BackgroundTransparency = 0.1 })
    end
    for i, tk in ipairs(Ticks) do
        local base = ((i - 1) % 5 == 0) and 0.15 or 0.6
        task.delay(i * 0.012, function()
            tween(tk, 0.25, ES.Quad, ED.Out, { BackgroundTransparency = base })
        end)
    end
    task.delay(1.1, function() ticksLive = true end)
    task.wait(0.9)

    for i, g in ipairs(Glow) do
        TweenService:Create(g,
            TweenInfo.new(1.4 + i * 0.25, ES.Sine, ED.InOut, -1, true),
            { BackgroundTransparency = glowTargets[i] + 0.05 }):Play()
    end
    TweenService:Create(LogoScale,
        TweenInfo.new(1.6, ES.Sine, ED.InOut, -1, true), { Scale = 1.04 }):Play()

    -- Phase 3: title decodes letter by letter, subtitle in, shimmer sweep
    for i, l in ipairs(Letters) do
        decodeLetter(l, (TITLE):sub(i, i), 0.3 + (i - 1) * 0.2)
    end
    task.wait(1.25)
    Subtitle.Position = UDim2.fromOffset(CX, 390)
    tween(Subtitle, 0.6, ES.Quad, ED.Out, { TextTransparency = 0.2, Position = UDim2.fromOffset(CX, 384) })
    task.wait(0.2)
    for i, l in ipairs(Letters) do
        task.delay((i - 1) * 0.09, function()
            l.grad.Offset = Vector2.new(-1, 0)
            tween(l.grad, 0.7, ES.Sine, ED.InOut, { Offset = Vector2.new(1, 0) })
        end)
    end
    task.wait(0.5)

    -- Phase 4: segmented bar + typed status (server warm-up decides the hold)
    for i, c in ipairs(Cells) do
        task.delay(i * 0.012, function()
            tween(c, 0.25, ES.Quad, ED.Out, { BackgroundTransparency = 0 })
        end)
    end
    tween(StageText, 0.4, ES.Quad, ED.Out, { TextTransparency = 0 })
    tween(PercentText, 0.4, ES.Quad, ED.Out, { TextTransparency = 0 })

    local function setProgress(p, text, dur)
        if text then typeStatus(text) end
        tween(ProgressValue, dur or 0.5, ES.Quad, ED.Out, { Value = p })
    end

    task.wait(0.35)
    setProgress(0.25, "Initializing...")
    task.wait(0.7)
    setProgress(0.55, "Checking environment...")
    task.wait(0.7)
    setProgress(0.8, "Connecting to server...", 0.7)

    local waited, wakeShown = 0, false
    while not warmDone and waited < 20 do
        task.wait(0.25)
        waited = waited + 0.25
        if waited > 3 and not wakeShown then
            wakeShown = true
            typeStatus("Waking up server...")
        end
    end

    if warmDone and warmOk then
        setProgress(1, "Ready")
    else
        setProgress(1, "Server not responding, continuing...")
    end
    task.wait(0.9)

    -- Phase 5: exit: outro sound hits, intro sound fades, visuals dissolve, key window slides in
    introRunning = false
    ticksLive = false
    fadeStopSfx(IntroSfx, INTRO_FADE_OUT)
    playSfx(OutroSfx, OUTRO_VOLUME, 0.05)
    flash(0.35, 0.5)
    shockwave(190, 620, 0.9, 2)
    tween(LogoScale, 0.6, ES.Back, ED.In, { Scale = 1.3 })
    fadeOutTree(Container, 0.6)
    tween(Container, 0.7, ES.Quad, ED.Out, { BackgroundTransparency = 1 })
    task.wait(0.5)

    KeyFrame.Visible = true
    fadeKey(true, 0.55)
    task.wait(1.2)
    beatLive = true


    for _, c in ipairs(introConns) do c:Disconnect() end
    Container:Destroy()
end

local function startHeartbeat(token, usedKey)
    task.spawn(function()
        local wasDisabled = false
        while true do
            task.wait(HEARTBEAT_EVERY)
            local res = request({
                Url = BACKEND_URL .. "/api/heartbeat",
                Method = "POST",
                Headers = { ["Content-Type"] = "application/json" },
                Body = HttpService:JSONEncode({ userId = LocalPlayer.UserId, token = token }),
            }, 1)
            local data = res and decode(res.Body)
            if data then
                if data.success then
                    wasDisabled = false
                    -- a new /update global message arrived while the script is running
                    if data.message_id and data.message_id ~= lastMsgId then
                        lastMsgId = data.message_id
                        if type(data.message) == "string" and data.message ~= "" then
                            notify("GENX", data.message)
                        end
                    end
                elseif data.reason == "disabled" then
                    -- staff paused the loader: tell the user once, keep pinging so it resumes
                    if not wasDisabled then notify("GENX", data.message or "Loader disabled by staff.") end
                    wasDisabled = true
                elseif data.reason then
                    -- key changed / revoked / expired / session gone: forget the saved key and stop
                    if data.forget and loadKey() == usedKey then forgetKey() end
                    pcall(function() getgenv().GENX_SESSION_ENDED = data.reason end)
                    notify("GENX", data.message or "Session ended.")
                    break
                end
                -- no reason (e.g. rate limited): just try again next beat
            end
        end
    end)
end

-- returns "ok" (script launched), "rejected" (server says this key is dead), or "fail" (anything else)
local function verifyKey(userKey, silent)
    if busy then return "fail" end
    userKey = (userKey or ""):gsub("%s+", "")
    if userKey == "" then
        setStatus("Please enter a key!", COLOR_ERR)
        shake()
        return "fail"
    end

    busy = true
    verifying = true
    task.spawn(function()
        local n = 0
        Spinner.Visible = true
        while verifying do
            EnterBtn.Text = "VERIFYING" .. string.rep(".", n % 4)
            n = n + 1
            task.wait(0.3)
        end
        Spinner.Visible = false
    end)
    setStatus("Verifying key...", COLOR_INFO)

    local outcome = "fail"
    local res, err = request({
        Url = BACKEND_URL .. "/api/verify",
        Method = "POST",
        Headers = { ["Content-Type"] = "application/json" },
        Body = HttpService:JSONEncode({
            loader_id = LOADER_ID, gameId = tostring(game.GameId),
            key = userKey,
            userId = LocalPlayer.UserId,
            username = LocalPlayer.Name,
            hwid = getHWID(),
        }),
    }, 8, function(i) setStatus("Waking up server, hang tight... (" .. i .. "/7)", COLOR_INFO) end)

    local data = res and decode(res.Body)

    if not data then
        setStatus("Server connection error!\n" .. tostring(err or "bad response"), COLOR_ERR)
        shake()
    elseif not data.success then
        setStatus(data.message or "Invalid key or loader disabled!", COLOR_ERR)
        if not silent then shake() end
        -- server says this key is dead (changed / revoked / expired / invalid):
        -- delete it from this device, but only if it's the one we had saved
        if data.forget then
            outcome = "rejected"
            if loadKey() == userKey then
                forgetKey()
                KeyBox.Text = ""
            end
        end
    else
        -- Compile first so a bad script never leaves the user with a dead UI
        local fn, compileErr
        if data.script and data.script ~= "" then
            fn, compileErr = loadstring(data.script)
        end

        if data.script and data.script ~= "" and not fn then
            setStatus("Script failed to load: " .. tostring(compileErr), COLOR_ERR)
        else
            saveKey(userKey)
            applyMessage(data.message)
            if quickMode and type(data.message) == "string" and data.message ~= "" then
                notify("GENX", data.message) -- quick login skips the intro, so show the global message as a toast
            end
            setStatus("Key valid! Loading script...", COLOR_OK)
            verifying = false
            EnterBtn.Text = "ACCESS GRANTED"
            fadeStopSfx(OutroSfx, 0.8) -- heartbeat stops now that the key is right
            task.wait(0.8)

            beatLive = false
            fadeKey(false, 0.4)
            ecgTarget = 0
            task.wait(0.5)
            ScreenGui:Destroy()

            if data.message_id then lastMsgId = data.message_id end
            if data.token then startHeartbeat(data.token, userKey) end
            if fn then
                local ok, runErr = pcall(fn)
                if not ok then warn("[GENX] Script error: " .. tostring(runErr)) end
            end
            return "ok"
        end
    end

    verifying = false
    EnterBtn.Text = "ENTER KEY"
    busy = false
    return outcome
end

local saved = loadKey()
if saved ~= "" then
    -- QUICK PATH: this device already has a saved key.
    -- Skip the big intro, show ONLY the key window, log in automatically, run the script.
    quickMode = true
    Container.Visible = false
    KeyBox.Text = saved
    setStatus("Saved key found, logging in...", COLOR_INFO)
    KeyFrame.Visible = true
    fadeKey(true, 0.45)
    task.delay(0.6, function() beatLive = true end)

    local result = verifyKey(saved, true)
    if result == "rejected" then
        -- Saved key is wrong now (changed / revoked / expired): it was already deleted from this
        -- device. Play the FULL animation, then land on the key window to enter the new key.
        quickMode = false
        beatLive = false
        fadeKey(false, 0.35)
        task.wait(0.4)
        KeyFrame.Visible = false
        Container.Visible = true
        playIntroAnimation()
    end
else
    -- First run / no saved key: full intro animation, then the key window
    playIntroAnimation()
end
